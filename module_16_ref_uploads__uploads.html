Загрузка файлов (File uploads)

Настройки размещения файлов

# settings.py
MEDIA_ROOT = os.path.join(BASE_DIR, 'media')   # где ФИЗИЧЕСКИ лежат файлы
MEDIA_URL = '/media/'                          # URL для отдачи

Раздача в dev (main urls.py):

from django.conf import settings
from django.conf.urls.static import static

urlpatterns += static(settings.MEDIA_URL, document_root=settings.MEDIA_ROOT)

Поле в модели

from django.db import models

class Document(models.Model):
    title = models.CharField(max_length=120)
    # upload_to — подпапка (можно функцию от instance)
    file = models.FileField(upload_to='docs/')
    image = models.ImageField(upload_to='imgs/', blank=True, null=True)
    uploaded_at = models.DateTimeField(auto_now_add=True)
  • ImageField требует Pillow: pip install Pillow.
  • имя файла уникализируется автоматически (добавляется суффикс).

Функция для upload_to:

def user_dir(instance, filename):
    return f'users/{instance.user.id}/{filename}'

class Avatar(models.Model):
    user = models.OneToOneField(User, on_delete=models.CASCADE)
    img = models.ImageField(upload_to=user_dir)

Форма с файлом

<!-- шаблон -->
<form method="post" enctype="multipart/form-data">
    {% csrf_token %}
    {{ form.as_p }}
    <button type="submit">Загрузить</button>
</form>
# forms.py
from django import forms
from .models import Document

class DocumentForm(forms.ModelForm):
    class Meta:
        model = Document
        fields = ['title', 'file']

# views.py
def upload_view(request):
    if request.method == 'POST':
        form = DocumentForm(request.POST, request.FILES)   # ОБЯЗАТЕЛЬНО request.FILES
        if form.is_valid():
            doc = form.save()          # файл сохранён в MEDIA_ROOT автоматически
            return redirect('success')
    else:
        form = DocumentForm()
    return render(request, 'upload.html', {'form': form})

Забыли request.FILES — файл не сохранится, форма не валидна по этому полю.

Обработка файла программно

def raw_upload(request):
    f = request.FILES['myfile']        # UploadedFile
    print(f.name, f.size)

    with open(f.name, 'wb+') as dest:  # медленно, для примеров
        for chunk in f.chunks():       # chunks() — читает по кускам
            dest.write(chunk)

Проверка размера, типа и расширения

MAX_SIZE = 2 * 1024 * 1024  # 2 МБ

def clean_file(self):
    f = self.cleaned_data['file']
    if f.size > MAX_SIZE:
        raise forms.ValidationError('Файл слишком большой')
    if not f.content_type.startswith('image/'):
        raise forms.ValidationError('Нужно изображение')
    ext = os.path.splitext(f.name)[1].lower()
    allowed = {'.pdf', '.doc', '.docx'}
    if ext not in allowed:
        raise forms.ValidationError('Недопустимое расширение')
    return f

Доступ к файлу в шаблоне

<p><a href="{{ doc.file.url }}">Скачать {{ doc.title }}</a></p>
<!-- doc.file.url == MEDIA_URL + относительный путь -->

Хранилища

from django.core.files.storage import default_storage
default_storage.save('x.txt', content)          # в MEDIA_ROOT
default_storage.exists('x.txt')
default_storage.delete('x.txt')

Для облаков (S3/MinIO) — сторонние django-storages (классы S3Boto3Storage и пр.);
в settings меняется DEFAULT_FILE_STORAGE (в Django ≥ 4.2 – STORAGES).

Файл при редактировании ворса

  • FileField не требует обязательной загрузки при правке — если поле пустое, сохраняется старый файл.
  • Для полной замены файла — загрузить новый; старый файл сам не удалится (удаляйте вручную при желании).
  • Пиксельные (ImageField) моменты: .width/.height доступны через .image (PIL).
